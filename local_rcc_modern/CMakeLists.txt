cmake_minimum_required(VERSION 3.24)
project(local_rcc_modern VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Dependencies: pass existing checkouts with -DLUAU_DIR=... -DZSTD_DIR=... -DXXHASH_DIR=...,
# otherwise they are fetched at these pinned versions.
include(FetchContent)
set(LUAU_TAG 0.744 CACHE STRING "Luau release")
if(NOT LUAU_DIR)
  FetchContent_Declare(luau GIT_REPOSITORY https://github.com/luau-lang/luau.git GIT_TAG ${LUAU_TAG})
  FetchContent_Populate(luau)
  set(LUAU_DIR ${luau_SOURCE_DIR})
endif()
if(NOT ZSTD_DIR)
  FetchContent_Declare(zstd GIT_REPOSITORY https://github.com/facebook/zstd.git GIT_TAG v1.5.6)
  FetchContent_Populate(zstd)
  set(ZSTD_DIR ${zstd_SOURCE_DIR})
endif()
if(NOT XXHASH_DIR)
  FetchContent_Declare(xxhash GIT_REPOSITORY https://github.com/Cyan4973/xxHash.git GIT_TAG v0.8.2)
  FetchContent_Populate(xxhash)
  set(XXHASH_DIR ${xxhash_SOURCE_DIR})
endif()

# Luau compiler (Ast + Compiler + Bytecode/Common headers), as a static library.
file(GLOB LUAU_SRC ${LUAU_DIR}/Ast/src/*.cpp ${LUAU_DIR}/Compiler/src/*.cpp ${LUAU_DIR}/Common/src/*.cpp ${LUAU_DIR}/Bytecode/src/*.cpp)
add_library(lrm_luau STATIC ${LUAU_SRC})
target_include_directories(lrm_luau PUBLIC ${LUAU_DIR}/Ast/include ${LUAU_DIR}/Compiler/include ${LUAU_DIR}/Common/include ${LUAU_DIR}/Bytecode/include)
set_target_properties(lrm_luau PROPERTIES POSITION_INDEPENDENT_CODE ON)

# zstd (compression and decompression only).
file(GLOB ZSTD_SRC ${ZSTD_DIR}/lib/common/*.c ${ZSTD_DIR}/lib/compress/*.c ${ZSTD_DIR}/lib/decompress/*.c)
add_library(lrm_zstd STATIC ${ZSTD_SRC})
target_include_directories(lrm_zstd PUBLIC ${ZSTD_DIR}/lib)
target_compile_definitions(lrm_zstd PRIVATE ZSTD_DISABLE_ASM ZSTD_MULTITHREAD=0)
set_target_properties(lrm_zstd PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(lrm_encoder STATIC src/encoder.cpp)
target_include_directories(lrm_encoder PUBLIC ${XXHASH_DIR})
target_compile_definitions(lrm_encoder PRIVATE XXH_INLINE_ALL)
target_link_libraries(lrm_encoder PUBLIC lrm_luau lrm_zstd)
set_target_properties(lrm_encoder PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(WIN32)
  add_library(local_rcc_modern SHARED src/dllmain.cpp)
  target_link_libraries(local_rcc_modern PRIVATE lrm_encoder)
  add_executable(local_rcc_inject src/injector.cpp)
  if(MINGW)
    target_link_options(local_rcc_modern PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(local_rcc_inject PRIVATE -static -municode)
  endif()
  set_target_properties(local_rcc_modern PROPERTIES PREFIX "")
else()
  enable_testing()
  add_executable(encoder_test tests/encoder_test.cpp)
  target_link_libraries(encoder_test PRIVATE lrm_encoder)
  add_test(NAME encoder COMMAND encoder_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sample.lua)
endif()
